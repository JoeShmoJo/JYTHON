############################
# Diversion name probe -- a TEMPORARY DIAGNOSTIC, not an operating rule.
#
# Purpose: find out how a scripted rule sitting on a Diversion can discover
# which diversion it is attached to. Reservoir rules use
# currentRule.getReservoirElement(); it is not known whether that works on a
# diversion, so this asks the objects directly instead of guessing.
#
# How to use it:
#   1. Paste this whole file into the scripted rule at the diversion
#      (the "script_test" rule in the Script Operation Rule editor).
#   2. Run a SHORT compute -- a few days is plenty.
#   3. Read the lines beginning "PROBE:" in the compute log and send them back.
#
# runRuleScript returns None, so this rule has no effect on the compute.
# Everything is wrapped in try/except, so a method that does not exist reports
# itself and the probe keeps going rather than halting the run.
############################
import sys

from hec.script import Constants

################################################################################
# FUNCTION DEFINITIONS

def _say(network, msg):
    """Write one line to the compute log. Never raises."""
    try:
        network.printMessage("PROBE: %s" % msg)
    except:
        pass


def _try(network, label, func):
    """
    Call func() and report what came back, or how it failed.
    Returns the value, or None if it failed. Never raises.
    """
    try:
        value = func()
        _say(network, "  %-44s -> %s" % (label, value))
        return value
    except:
        _say(network, "  %-44s -> FAILED: %s" % (label, sys.exc_info()[1]))
        return None


def _listMethods(network, obj, label):
    """Print the methods on obj whose names suggest they identify an element."""
    interesting = []
    everything = []
    try:
        for name in dir(obj):
            if name.startswith("__"):
                continue
            everything.append(name)
            low = name.lower()
            for hint in ["elem", "name", "divers", "parent", "owner",
                         "system", "control", "operat"]:
                if hint in low:
                    interesting.append(name)
                    break
    except:
        _say(network, "  dir(%s) FAILED: %s" % (label, sys.exc_info()[1]))
        return
    interesting.sort()
    everything.sort()
    _say(network, "  likely-relevant methods on %s:" % label)
    _say(network, "    %s" % ", ".join(interesting))
    # Full list too, in chunks, so nothing is missed
    _say(network, "  all %d members of %s:" % (len(everything), label))
    chunk = 8
    for i in range(0, len(everything), chunk):
        _say(network, "    %s" % ", ".join(everything[i:i + chunk]))


def _findOwningDiversion(currentRule, network):
    """
    Work backwards: walk the diversions in the network, ask each one for the
    rules it holds, and see which list contains this rule. Uses only calls
    already proven in this repo -- see cResSim.getDiversionTSC, which does
    divElem.getController().getRuleVector()[0].
    """
    _say(network, "reverse lookup: which diversion holds this rule?")
    divNames = _try(network, "network.getDiversionNames()",
                    lambda: list(network.getDiversionNames()))
    if not divNames:
        _say(network, "  no diversion name list, so nothing to walk")
        return None
    myName = None
    try:
        myName = currentRule.getName()
    except:
        pass
    for divName in divNames:
        divElem = None
        try:
            divElem = network.findDiversion(divName)
        except:
            try:
                divElem = network.findElement(divName)
            except:
                divElem = None
        if divElem is None:
            _say(network, "  %-30s could not be looked up" % divName)
            continue
        try:
            rules = list(divElem.getController().getRuleVector())
        except:
            _say(network, "  %-30s getController().getRuleVector() FAILED: %s"
                 % (divName, sys.exc_info()[1]))
            continue
        ruleNames = []
        matched = False
        for rule in rules:
            thisName = None
            try:
                thisName = rule.getName()
            except:
                pass
            ruleNames.append(str(thisName))
            if rule is currentRule:
                matched = True
            elif myName is not None and thisName == myName:
                matched = True
        _say(network, "  %-30s holds rules: %s%s"
             % (divName, ", ".join(ruleNames), "   <== MATCH" if matched else ""))
        if matched:
            return divName
    return None


def initRuleScript(currentRule, network):
    """Everything happens here, so it runs once per compute rather than per step."""
    try:
        _say(network, "=" * 70)
        _say(network, "DIVERSION NAME PROBE")

        _say(network, "the rule's own identity")
        _try(network, "currentRule.getName()", lambda: currentRule.getName())
        _try(network, "currentRule.getDescription()",
             lambda: currentRule.getDescription())
        _try(network, "currentRule.toString()", lambda: currentRule.toString())

        _say(network, "the method reservoir rules use for this")
        elem = _try(network, "currentRule.getReservoirElement()",
                    lambda: currentRule.getReservoirElement())
        if elem is not None:
            _try(network, "  element.toString()", lambda: elem.toString())
            _try(network, "  element.getName()", lambda: elem.getName())
            _try(network, "  element._name", lambda: elem._name)

        _say(network, "other ways the rule might reach its element")
        _try(network, "currentRule.getSystem()", lambda: currentRule.getSystem())
        _try(network, "currentRule.getController()",
             lambda: currentRule.getController())
        _try(network, "currentRule.getElement()", lambda: currentRule.getElement())
        _try(network, "currentRule.getOperationsElement()",
             lambda: currentRule.getOperationsElement())
        _try(network, "currentRule.getDiversionElement()",
             lambda: currentRule.getDiversionElement())

        _listMethods(network, currentRule, "currentRule")

        divName = _findOwningDiversion(currentRule, network)
        _say(network, "ANSWER: owning diversion = %s" % divName)
        _say(network, "=" * 70)
    except:
        _say(network, "probe itself failed: %s" % sys.exc_info()[1])
    # Always succeed, so a failed probe never halts the compute
    return Constants.TRUE


def runRuleScript(currentRule, network, currentRuntimestep):
    # Returning None means this rule has no effect on the compute.
    return None
