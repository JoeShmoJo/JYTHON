# MinFlowPlusWithdrawal scripted rule
#
# Paste this into the ResSim scripted rule editor. All the real code lives in
# scripts/externalRules/MinFlowPlusWithdrawal.py, so it exists in one place
# instead of being copied into every reservoir.
#
# The reload() below is the important part. ResSim keeps imported Jython modules
# alive for the whole session, so without it, edits to the .py are not picked up
# until ResSim is closed and reopened. Same pattern as
# scripts/Willamette/StateVars/PctFull_LOP_Init.py
############################
import sys

def initRuleScript(currentRule, network):
    #Make sure the scripts folder is importable, then load the external module
    modulePath = network.makeAbsolutePathFromWatershed("scripts")
    if not modulePath in sys.path:
        sys.path.append(modulePath)
    from externalRules import MinFlowPlusWithdrawal
    reload(MinFlowPlusWithdrawal)
    return MinFlowPlusWithdrawal.initRuleScript(currentRule, network)

def runRuleScript(currentRule, network, currentRuntimestep):
    from externalRules import MinFlowPlusWithdrawal
    return MinFlowPlusWithdrawal.runRuleScript(currentRule, network, currentRuntimestep)
